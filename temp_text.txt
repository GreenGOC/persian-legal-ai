document = LegalDocument.objects.get(id=14583)

elements = LegalElement.objects.filter(
    document=document
).select_related(
    "provision",
    "structural",
).order_by("order", "id")

for element in elements:
    if element.element_type == ElementType.PROVISION:
        print(
            "PROVISION:",
            element.provision.provision_type,
            element.provision.number,
			element.provision.title,
            element.provision.text[:100],
        )
    elif element.element_type == ElementType.STRUCTURAL:
        print(
            "STRUCTURAL:",
            element.structural.structural_type,
            element.structural.number,
            element.structural.title,
        )
